import os
import json
import pygame

class PlayerRegistration:
    def __init__(self, filepath="PlayerProfile.txt"):
        self.filepath = filepath
        self.name = ""
        self.score = 0
        self.rupees = 0
        self.completed_questions = set()
        self.load_profile()
        
        # UI State for Registration Screen
        self.input_text = self.name
        self.active = True
        self.font = pygame.font.SysFont(None, 32)
        self.is_registered = bool(self.name)

    def handle_event(self, event):
        if not self.active:
            return
            
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_RETURN:
                if self.input_text.strip():
                    self.register_player(self.input_text)
                    self.active = False
            elif event.key == pygame.K_BACKSPACE:
                self.input_text = self.input_text[:-1]
            else:
                if len(self.input_text) < 20:  # Character limit
                    self.input_text += event.unicode

    def register_player(self, name):
        self.name = name.strip()
        self.is_registered = True
        self.save_profile()

    def is_phase_complete(self, phase_number, total_phase_questions=10):
        phase_questions = [q for q in self.completed_questions if str(q).startswith(f"p{phase_number}_")]
        return len(phase_questions) >= total_phase_questions

    def add_score_and_rupees(self, points=1, rupees=10):
        self.score += points
        self.rupees += rupees
        self.save_profile()

    def save_profile(self):
        data = {
            "name": self.name,
            "score": self.score,
            "rupees": self.rupees,
            "completed_questions": list(self.completed_questions)
        }
        try:
            with open(self.filepath, "w") as f:
                json.dump(data, f, indent=4)
        except Exception as e:
            print(f"Error saving profile: {e}")

    def load_profile(self):
        if os.path.exists(self.filepath):
            try:
                with open(self.filepath, "r") as f:
                    data = json.load(f)
                    self.name = data.get("name", "")
                    self.score = data.get("score", 0)
                    self.rupees = data.get("rupees", 0)
                    self.completed_questions = set(data.get("completed_questions", []))
            except Exception as e:
                print(f"Error loading profile: {e}")

    def draw(self, surface):
        if not self.active and self.is_registered:
            return  # Registration overlay hidden once completed

        # Draw dark background overlay
        surface.fill((20, 20, 30), pygame.Rect(150, 150, 500, 200))
        pygame.draw.rect(surface, (100, 100, 150), pygame.Rect(150, 150, 500, 200), 2)

        # Title
        title_surf = self.font.render("Enter Player Name:", True, (255, 255, 255))
        surface.blit(title_surf, (180, 180))

        # Input Box
        box_rect = pygame.Rect(180, 230, 440, 40)
        pygame.draw.rect(surface, (50, 50, 70), box_rect)
        pygame.draw.rect(surface, (0, 150, 255), box_rect, 2)

        text_surf = self.font.render(self.input_text, True, (255, 255, 255))
        surface.blit(text_surf, (box_rect.x + 10, box_rect.y + 8))

        # Instruction
        inst_surf = pygame.font.SysFont(None, 20).render("Press ENTER to Save & Continue", True, (180, 180, 180))
        surface.blit(inst_surf, (180, 285))
