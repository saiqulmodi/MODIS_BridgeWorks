import pygame
from game.questions import QuestionManager

class App:
    def __init__(self):
        self.screen = pygame.display.set_mode((800, 600))
        pygame.display.set_caption("MODIS_BridgeWorks Academy")
        self.clock = pygame.time.Clock()
        self.running = True
        
        # Initialize fonts and the Question Manager
        self.font_large = pygame.font.SysFont(None, 36)
        self.font_medium = pygame.font.SysFont(None, 28)
        self.q_manager = QuestionManager()
        
        # UI Theme Colors
        self.COLOR_BG = (15, 23, 42)          # Dark blue
        self.COLOR_CARD = (30, 41, 59)        # Card background
        self.COLOR_TEXT = (255, 255, 255)     # White text
        self.COLOR_ACCENT = (255, 215, 0)     # Gold for scores/grants
        self.COLOR_OPTION = (51, 65, 85)      # Option button fill

        # Player stats
        self.civil_grants = 28000
        self.feedback_message = ""

    async def run(self):
        while self.running:
            self.handle_events()
            self.update()
            self.render()
            await asyncio.sleep(0)  # Required for pygbag async browser loop
        pygame.quit()

    def handle_events(self):
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                self.running = False
            elif event.type == pygame.MOUSEBUTTONDOWN:
                # Check option clicks
                mouse_pos = pygame.mouse.get_pos()
                q_data = self.q_manager.get_current_question()
                
                if q_data and hasattr(self, 'option_rects'):
                    for rect, option_text in self.option_rects:
                        if rect.collidepoint(mouse_pos):
                            is_correct = self.q_manager.check_answer(option_text)
                            if is_correct:
                                self.civil_grants += 7000
                                self.feedback_message = "Correct! +Rs 7,000 Civil Grant"
                            else:
                                self.feedback_message = "Incorrect, try again!"

    def update(self):
        pass  # Add game physics or timer updates here if needed

    def render(self):
        self.screen.fill(self.COLOR_BG)

        # 1. Render Header / Civil Grants
        grants_surf = self.font_large.render(f"Civil Grants: Rs {self.civil_grants:,}", True, self.COLOR_ACCENT)
        self.screen.blit(grants_surf, (50, 30))

        # 2. Draw Question Card Area
        card_rect = pygame.Rect(50, 100, 700, 440)
        pygame.draw.rect(self.screen, self.COLOR_CARD, card_rect, border_radius=12)

        q_data = self.q_manager.get_current_question()
        self.option_rects = []

        if q_data:
            # Render Question Text
            q_surf = self.font_medium.render(q_data["question"], True, self.COLOR_TEXT)
            self.screen.blit(q_surf, (80, 130))

            # Render Options
            start_y = 200
            for i, option in enumerate(q_data.get("options", [])):
                opt_rect = pygame.Rect(80, start_y + (i * 65), 640, 50)
                pygame.draw.rect(self.screen, self.COLOR_OPTION, opt_rect, border_radius=8)

                prefix = f"{chr(65 + i)}. "
                opt_surf = self.font_medium.render(prefix + option, True, self.COLOR_TEXT)
                self.screen.blit(opt_surf, (100, start_y + (i * 65) + 12))
                
                self.option_rects.append((opt_rect, option))
        else:
            done_surf = self.font_medium.render("All questions completed for this subject!", True, (0, 255, 0))
            self.screen.blit(done_surf, (80, 150))

        # Render Feedback Message
        if self.feedback_message:
            feedback_surf = self.font_medium.render(self.feedback_message, True, self.COLOR_ACCENT)
            self.screen.blit(feedback_surf, (80, 480))

        pygame.display.flip()
        self.clock.tick(60)
